Multipart File Upload

Upload files directly without Base64 encoding NEW 30% FASTER v1

Why Multipart?

✅ Преимущества multipart/form-data:
• Меньше размер - без Base64 overhead (~33% экономии)
• Быстрее загрузка - нативная поддержка браузером
• Простота - не нужно конвертировать в Base64
• Большие файлы - до 100 MB (vs 10 MB для Base64)

❌ Старый способ (Base64)

Проблемы:
  • Файл 10 MB → 13.3 MB после Base64
  • Дополнительное время на кодирование
  • Нагрузка на память

✅ Новый способ (Multipart)

Плюсы:
  • Файл 10 MB остаётся 10 MB
  • Прямая загрузка
  • Меньше памяти

Endpoints

Endpoint Content-Type File Format Max Size
/api/v1/ai/process application/json Base64 (legacy) 10 MB
/api/v1/ai/process multipart/form-data Binary (new) 100 MB
/api/v1/ai/stream multipart/form-data Binary (new) 100 MB
🔄 Обратная совместимость:
API автоматически определяет формат по Content-Type. Старые клиенты с Base64 продолжат работать без изменений.

Client Examples

JavaScript (Browser - FormData)

async function uploadWithFiles() {
  const formData = new FormData();
  
  // Add text fields
  formData.append('provider', 'gemini');
  formData.append('model', 'gemini-1.5-pro');
  formData.append('prompt', 'Analyze this image');
  
  // Add parameters as JSON string
  formData.append('parameters', JSON.stringify({
    temperature: 0.7,
    max_tokens: 1024
  }));
  
  // Add files from file input
  const fileInput = document.getElementById('fileInput');
  for (const file of fileInput.files) {
    formData.append('uploaded_files[]', file);
  }
  
  // Send request
  const response = await fetch('https://api.siteaccess.ru/api/v1/ai/process', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${token}`
      // Do NOT set Content-Type manually - browser will set it with boundary
    },
    body: formData
  });
  
  const result = await response.json();
  console.log(result);
}

JavaScript (Node.js - form-data)

const FormData = require('form-data');
const fs = require('fs');
const axios = require('axios');

async function uploadWithFiles() {
  const form = new FormData();
  
  // Add text fields
  form.append('provider', 'gemini');
  form.append('model', 'gemini-1.5-pro');
  form.append('prompt', 'Analyze this image');
  
  // Add parameters
  form.append('parameters', JSON.stringify({
    temperature: 0.7,
    max_tokens: 1024
  }));
  
  // Add files
  form.append('uploaded_files[]', fs.createReadStream('photo.jpg'));
  form.append('uploaded_files[]', fs.createReadStream('document.pdf'));
  
  // Send request
  const response = await axios.post(
    'https://api.siteaccess.ru/api/v1/ai/process',
    form,
    {
      headers: {
        'Authorization': `Bearer ${token}`,
        ...form.getHeaders() // Important: includes Content-Type with boundary
      }
    }
  );
  
  console.log(response.data);
}

Python (requests)

import requests

def upload_with_files():
    url = 'https://api.siteaccess.ru/api/v1/ai/process'
    
    headers = {
        'Authorization': f'Bearer {token}'
        # Do NOT set Content-Type - requests will set it automatically
    }
    
    # Form data
    data = {
        'provider': 'gemini',
        'model': 'gemini-1.5-pro',
        'prompt': 'Analyze this image',
        'parameters': '{"temperature": 0.7, "max_tokens": 1024}'
    }
    
    # Files to upload
    files = [
        ('uploaded_files[]', ('photo.jpg', open('photo.jpg', 'rb'), 'image/jpeg')),
        ('uploaded_files[]', ('document.pdf', open('document.pdf', 'rb'), 'application/pdf'))
    ]
    
    response = requests.post(url, headers=headers, data=data, files=files)
    result = response.json()
    
    print(result)

PHP

<?php
$token = 'your-bearer-token';
$url = 'https://api.siteaccess.ru/api/v1/ai/process';

$ch = curl_init();

$data = [
    'provider' => 'gemini',

    CURLOPT_URL => $url,
    CURLOPT_POST => true,
    CURLOPT_POSTFIELDS => $data,
    CURLOPT_RETURNTRANSFER => true,
    CURLOPT_HTTPHEADER => [
        "Authorization: Bearer $token"
    ]
]);

$response = curl_exec($ch);
$result = json_decode($response, true);

print_r($result);

curl_close($ch);
?>

Best Practices

✅ Рекомендации:
1. Не устанавливайте Content-Type вручную - пусть библиотека добавит boundary
2. Оптимизируйте изображения перед загрузкой (resize, compression)
3. Используйте streaming для больших файлов (/api/v1/ai/stream)
4. Обрабатывайте ошибки загрузки - network может прерваться
5. Показывайте прогресс пользователю при больших файлах

Progress Tracking (JavaScript)

async function uploadWithProgress() {
  const formData = new FormData();
  // ... add fields and files
  
  const xhr = new XMLHttpRequest();
  
  // Track upload progress
  xhr.upload.addEventListener('progress', (e) => {
    if (e.lengthComputable) {
      const percentComplete = (e.loaded / e.total) * 100;
      console.log(`Upload: ${percentComplete}%`);
      updateProgressBar(percentComplete);
    }
  });
  
  // Handle completion
  xhr.addEventListener('load', () => {
    const result = JSON.parse(xhr.responseText);
    console.log('Upload complete:', result);
  });
  
  xhr.open('POST', 'https://api.siteaccess.ru/api/v1/ai/process');
  xhr.setRequestHeader('Authorization', `Bearer ${token}`);
  xhr.send(formData);
}

Migration from Base64

Before (Base64)

// Old way - Base64
const fileInput = document.getElementById('file');
const file = fileInput.files[0];

const reader = new FileReader();
reader.onload = async (e) => {
  const base64 = e.target.result.split(',')[1];
  
  const response = await fetch('/api/v1/ai/process', {
    method: 'POST',
    headers: {
      'Authorization': `Bearer ${token}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      provider: 'gemini',
      model: 'gemini-1.5-pro',
      prompt: 'Analyze this image',
      files: [{
        type: 'image',
        content: `data:image/jpeg;base64,${base64}`,
        mime_type: 'image/jpeg',
        name: file.name
      }]
    })
  });
};
reader.readAsDataURL(file);

After (Multipart)

// New way - Multipart (simpler!)
const fileInput = document.getElementById('file');
const formData = new FormData();

formData.append('provider', 'gemini');
formData.append('model', 'gemini-1.5-pro');
formData.append('prompt', 'Analyze this image');
formData.append('uploaded_files[]', fileInput.files[0]);

const response = await fetch('/api/v1/ai/process', {
  method: 'POST',
  headers: {
    'Authorization': `Bearer ${token}`
  },
  body: formData
});
⚠️ Важно:
• Base64 метод НЕ удаляется - он остаётся для обратной совместимости
• Старые клиенты продолжат работать без изменений
• Multipart рекомендуется для новых интеграций
